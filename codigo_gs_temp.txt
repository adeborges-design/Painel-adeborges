// Campos "* Busca" (29/ago/2026): o que o PRETENDENTE está procurando, não o imóvel do
// atendimento em si — "Tipo Busca" (Locação/Compra), "Categoria Busca" (Residencial/
// Comercial), "Tipo Imovel Busca" e "Bairro Busca". Preenchidos no formulário de
// atendimento (index.html) e usados pelo botão "🔍 Ver imóveis compatíveis" pra abrir a
// lista de Imóveis Disponíveis já filtrada com o perfil de busca desse cliente.

function garantirAbaContas_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(ABA_CONTAS);
  if (sheet && sheet.getLastColumn() < HEADERS_CONTAS.length) {
    sheet.getRange(1, 1, 1, HEADERS_CONTAS.length).setValues([HEADERS_CONTAS]);
    sheet.getRange(1, 1, 1, HEADERS_CONTAS.length).setBackground('#0d1f35').setFontColor('#c8a96e').setFontWeight('bold');
  }
  if (!sheet) {
    sheet = ss.insertSheet(ABA_CONTAS);
    sheet.getRange(1, 1, 1, HEADERS_CONTAS.length).setValues([HEADERS_CONTAS]);
    sheet.getRange(1, 1, 1, HEADERS_CONTAS.length)
      .setBackground('#0d1f35').setFontColor('#c8a96e').setFontWeight('bold');
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function numBr_(v) {
  v = String(v == null ? '' : v).trim();
  if (!v) return 0;
  if (v.indexOf(',') >= 0) { v = v.replace(/\./g, '').replace(',', '.'); }
  const n = parseFloat(v);
  return isNaN(n) ? 0 : n;
}

function garantirAbaGen_(nome, headers) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(nome);
  if (!sheet) {
    sheet = ss.insertSheet(nome);
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    sheet.getRange(1, 1, 1, headers.length)
      .setBackground('#0d1f35').setFontColor('#c8a96e').setFontWeight('bold');
    sheet.setFrozenRows(1);
    return sheet;
  }
  if (sheet.getLastColumn() < headers.length) {
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    sheet.getRange(1, 1, 1, headers.length)
      .setBackground('#0d1f35').setFontColor('#c8a96e').setFontWeight('bold');
    return sheet;
  }
  // Mesmo com a quantidade de colunas certa, o TEXTO do cabecalho pode estar desatualizado
  // (ex.: uma coluna foi renomeada no codigo depois que a aba ja tinha sido criada uma vez).
  // Sem essa sincronizacao, a leitura via nome de coluna pega o valor da posicao errada.
  const cabecalhoAtual = sheet.getRange(1, 1, 1, headers.length).getValues()[0].map(function(h){ return String(h).trim(); });
  const precisaAtualizar = headers.some(function(h, i){ return cabecalhoAtual[i] !== h; });
  if (precisaAtualizar) {
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    sheet.getRange(1, 1, 1, headers.length)
      .setBackground('#0d1f35').setFontColor('#c8a96e').setFontWeight('bold');
  }
  return sheet;
}

// Decide quais itens de "Outras Cobrancas" continuam pro mes seguinte, ao gerar um novo
// mes em Locacoes: "mensal" continua igual, "parcelado" avanca uma parcela e para sozinho
// quando chega na ultima, "pontual" (ou formato antigo, sem tipo) nao continua. O "destino"
// (repasse ao locador ou so entre escritorio/locatario) e o "incide na taxa de administracao"
// sao preservados ao continuar.
// Mesmo formato usado no frontend: "descricao::valor::tipo::parcelaAtual::parcelaTotal::destino::incideTaxa||...".
function continuarOutrasCobrancas_(outrasDescAntiga) {
  var bruto = String(outrasDescAntiga || '').trim();
  if (!bruto || bruto.indexOf('::') < 0) return ''; // vazio ou formato antigo (sem tipo) - nao continua
  var itensNovos = [];
  bruto.split('||').forEach(function(par) {
    var partes = par.split('::');
    var desc = (partes[0] || '').trim();
    var valor = partes[1];
    var tipo = partes[2] || 'pontual';
    var parcelaAtual = parseInt(partes[3]) || 0;
    var parcelaTotal = parseInt(partes[4]) || 0;
    var destino = partes[5] || 'repasse';
    var incideTaxa = partes[6] || '0';
    if (!desc) return;
    if (tipo === 'mensal') {
      itensNovos.push(desc + '::' + valor + '::mensal::0::0::' + destino + '::' + incideTaxa);
    } else if (tipo === 'parcelado' && parcelaAtual < parcelaTotal) {
      itensNovos.push(desc + '::' + valor + '::parcelado::' + (parcelaAtual + 1) + '::' + parcelaTotal + '::' + destino + '::' + incideTaxa);
    }
    // tipo 'pontual' (ou parcelado ja na ultima parcela) nao entra - fica so naquele mes
  });
  return itensNovos.join('||');
}

function proximoId_(sheet) {
  const ultimaLinha = sheet.getLastRow();
  let proximoId = 1;
  if (ultimaLinha > 1) {
    const ids = sheet.getRange(2, 1, ultimaLinha - 1, 1).getValues()
      .flat().map(function(v){return parseInt(v);}).filter(function(v){return !isNaN(v) && v > 0;});
    if (ids.length > 0) proximoId = Math.max.apply(null, ids) + 1;
  }
  return proximoId;
}

function crudGen_(acao, sufixo, aba, headers, body) {
  if (acao === 'lancar_' + sufixo) {
    const lock = LockService.getScriptLock();
    lock.waitLock(15000); // espera ate 15s se outro pedido estiver gravando na mesma hora
    try {
      const sheet = garantirAbaGen_(aba, headers);
      const campos = body.campos || {};
      const id = proximoId_(sheet);
      const novaLinha = headers.map(function(h, idx) {
        if (idx === 0) return id;
        return campos[h] || '';
      });
      sheet.appendRow(novaLinha);
      SpreadsheetApp.flush();
      return json_({ ok: true, id: id });
    } finally {
      lock.releaseLock();
    }
  }
  if (acao === 'atualizar_' + sufixo) {
    const sheet = garantirAbaGen_(aba, headers);
    const data = sheet.getDataRange().getValues();
    const id = parseInt(body.id);
    let linha = -1;
    for (let i = 1; i < data.length; i++) {
      if (parseInt(data[i][0]) === id) { linha = i + 1; break; }
    }
    if (linha < 0) return json_({ ok: false, error: 'Registro nao encontrado' });
    const campos = body.campos || {};
    headers.forEach(function(h, idx) {
      if (idx === 0) return;
      if (campos.hasOwnProperty(h)) sheet.getRange(linha, idx + 1).setValue(campos[h]);
    });
    SpreadsheetApp.flush();
    return json_({ ok: true });
  }
  if (acao === 'excluir_' + sufixo) {
    const sheet = garantirAbaGen_(aba, headers);
    const data = sheet.getDataRange().getValues();
    const id = parseInt(body.id);
    let linha = -1;
    for (let i = 1; i < data.length; i++) {
      if (parseInt(data[i][0]) === id) { linha = i + 1; break; }
    }
    if (linha < 0) return json_({ ok: false, error: 'Registro nao encontrado' });
    sheet.deleteRow(linha);
    SpreadsheetApp.flush();
    return json_({ ok: true });
  }
  return null;
}

  function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents);
    const acao = body.acao;
    const ss = SpreadsheetApp.getActiveSpreadsheet();

    // PROCESSOS
    if (acao === 'atualizar') {
      const sheet = ss.getSheetByName(ABA_PROCESSOS);
      if (!sheet) return json_({ ok: false, error: 'Aba nao encontrada' });
      if (body.campos && Object.prototype.hasOwnProperty.call(body.campos, 'Chave de Acesso')) garantirColunaProcessos_('Chave de Acesso');
      if (body.campos && Object.prototype.hasOwnProperty.call(body.campos, 'Senha Cliente')) garantirColunaProcessos_('Senha Cliente');
      // Auto-cria as colunas novas de "Advogado da Parte Adversa", "Fatos e Providências" e os
      // alertas/badges (Idoso, Opção por Juízo 100% Digital, Data de Autuação e os 3 novos de
      // 01/set/2026: Antecipação de Tutela Deferida, Autor com Desinteresse na Conciliação,
      // Justiça Gratuita) na primeira vez que aparecem no corpo do pedido — mesmo padrão usado
      // acima para Chave de Acesso/Senha Cliente, evita o usuário ter que criar a coluna
      // manualmente na planilha.
      ['Advogado Adversário Nome','Advogado Adversário OAB','Advogado Adversário Telefone','Advogado Adversário Email','Narrativa dos Fatos','Providências e Observações ao Advogado',
       'Idoso','Opcao Juizo Digital','Data de Autuacao','Antecipacao Tutela Deferida','Autor Desinteresse Conciliacao','Justica Gratuita','Link Artefato',
       // Honorários por fórmula (02/set/2026): "% do valor a receber" ou "N benefícios + %".
       'Forma Calculo Honorarios','Percentual Honorarios','Numero Beneficios Honorarios','Valor Beneficio Honorarios','Valor a Receber Processo',
       // Novos alertas (04/set/2026): Petição Urgente, Intervenção do Ministério Público, Perícia.
       'Peticao Urgente','Intervencao Ministerio Publico','Pericia',
       // Segunda Instância (04/set/2026) e Reconvenção + Situação (09/set/2026).
       'Segundo Grau','Segundo Grau Data Remessa','Segundo Grau Recurso','Segundo Grau Numero','Segundo Grau Camara','Segundo Grau Relator','Segundo Grau Observacoes',
       'Reconvencao','Reconvencao Data','Reconvencao Status','Reconvencao Observacoes','Situacao'].forEach(function(nomeCol){
        if (body.campos && Object.prototype.hasOwnProperty.call(body.campos, nomeCol)) garantirColunaProcessos_(nomeCol);
      });
      const data = sheet.getDataRange().getValues();
      const headers = data[0].map(h => String(h).trim());
      const linha = parseInt(body.linha);
      if (!linha || linha < 2) return json_({ ok: false, error: 'Linha invalida' });
      const campos = body.campos;
      const headersN = headers.map(normHdr_);
      const colunasNaoEncontradas = [];
      Object.keys(campos).forEach(nomeCol => {
        const colIdx = headersN.indexOf(normHdr_(nomeCol));
        if (colIdx >= 0) sheet.getRange(linha, colIdx + 1).setValue(campos[nomeCol]);
        else colunasNaoEncontradas.push(nomeCol);
      });
      const dataAtuIdx = headersN.indexOf(normHdr_('Data Atualizacao'));
      if (dataAtuIdx >= 0) {
        sheet.getRange(linha, dataAtuIdx + 1).setValue(
          Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'dd/MM/yyyy')
        );
      }
      SpreadsheetApp.flush();
      return json_({ ok: true, acao: 'atualizar', linha: linha, colunasNaoEncontradas: colunasNaoEncontradas });
    }

    if (acao === 'cadastrar') {
      const sheet = ss.getSheetByName(ABA_PROCESSOS);
      if (!sheet) return json_({ ok: false, error: 'Aba nao encontrada' });
      if (body.campos && Object.prototype.hasOwnProperty.call(body.campos, 'Chave de Acesso')) garantirColunaProcessos_('Chave de Acesso');
      if (body.campos && Object.prototype.hasOwnProperty.call(body.campos, 'Senha Cliente')) garantirColunaProcessos_('Senha Cliente');
      // Mesma auto-criação de colunas de alerta que o "atualizar" já faz (ver comentário lá) —
      // sem isto, cadastrar um processo NOVO já marcando um alerta pela primeira vez perderia o
      // valor silenciosamente (colunaNaoEncontrada), até alguém salvar de novo via "editar".
      ['Idoso','Opcao Juizo Digital','Data de Autuacao','Antecipacao Tutela Deferida','Autor Desinteresse Conciliacao','Justica Gratuita','Link Artefato',
       'Forma Calculo Honorarios','Percentual Honorarios','Numero Beneficios Honorarios','Valor Beneficio Honorarios','Valor a Receber Processo',
       'Peticao Urgente','Intervencao Ministerio Publico','Pericia',
       'Segundo Grau','Segundo Grau Data Remessa','Segundo Grau Recurso','Segundo Grau Numero','Segundo Grau Camara','Segundo Grau Relator','Segundo Grau Observacoes',
       'Reconvencao','Reconvencao Data','Reconvencao Status','Reconvencao Observacoes','Situacao'].forEach(function(nomeCol){
        if (body.campos && Object.prototype.hasOwnProperty.call(body.campos, nomeCol)) garantirColunaProcessos_(nomeCol);
      });
      const data = sheet.getDataRange().getValues();
      const headers = data[0].map(h => String(h).trim());
      const campos = body.campos;
      const ultimaLinha = sheet.getLastRow();
      let proximoNum = 1;
      if (ultimaLinha > 1) {
        const nums = sheet.getRange(2, 1, ultimaLinha - 1, 1).getValues()
          .flat().map(v => parseInt(v)).filter(v => !isNaN(v) && v > 0);
        if (nums.length > 0) proximoNum = Math.max.apply(null, nums) + 1;
      }
      const headersN = headers.map(normHdr_);
      const camposN = {};
      Object.keys(campos).forEach(function(k){ camposN[normHdr_(k)] = campos[k]; });
      const colunasNaoEncontradas = Object.keys(campos).filter(function(k){ return headersN.indexOf(normHdr_(k)) < 0; });
      const novaLinha = headers.map(function(h, idx) {
        return idx === 0 ? proximoNum : (camposN[normHdr_(h)] || '');
      });
      var colB = sheet.getRange(2, 2, Math.max(sheet.getLastRow()-1,1), 1).getValues();
      var lastReal = 1;
      for (var i = colB.length - 1; i >= 0; i--) { if (String(colB[i][0]).trim()) { lastReal = i + 2; break; } }
      sheet.getRange(lastReal + 1, 1, 1, novaLinha.length).setValues([novaLinha]);
      SpreadsheetApp.flush();
      return json_({ ok: true, acao: 'cadastrar', nro: proximoNum, colunasNaoEncontradas: colunasNaoEncontradas });
    }

    // PORTAL DO CLIENTE - gerar/revogar o token de acesso somente-leitura de um processo.
    if (acao === 'gerar_token_processo') {
      const sheetTk = ss.getSheetByName(ABA_PROCESSOS);
      if (!sheetTk) return json_({ ok: false, error: 'Aba nao encontrada' });
      const linhaTk = parseInt(body.linha);
      if (!linhaTk || linhaTk < 2) return json_({ ok: false, error: 'Linha invalida' });
      const colTk = garantirColunaProcessos_(COL_TOKEN_CLIENTE);
      const tokenAtual = String(sheetTk.getRange(linhaTk, colTk).getValue() || '').trim();
      if (tokenAtual && !body.forcarNovo) {
        return json_({ ok: true, token: tokenAtual, novo: false });
      }
      const tokenNovo = Utilities.getUuid();
      sheetTk.getRange(linhaTk, colTk).setValue(tokenNovo);
      SpreadsheetApp.flush();
      return json_({ ok: true, token: tokenNovo, novo: true });
    }

    if (acao === 'revogar_token_processo') {
      const sheetTk2 = ss.getSheetByName(ABA_PROCESSOS);
      if (!sheetTk2) return json_({ ok: false, error: 'Aba nao encontrada' });
      const linhaTk2 = parseInt(body.linha);
      if (!linhaTk2 || linhaTk2 < 2) return json_({ ok: false, error: 'Linha invalida' });
      const colTk2 = garantirColunaProcessos_(COL_TOKEN_CLIENTE);
      sheetTk2.getRange(linhaTk2, colTk2).setValue('');
      SpreadsheetApp.flush();
      return json_({ ok: true });
    }

    // PORTAL DO CLIENTE - login por CPF/CNPJ + senha (alternativa ao link com token: o cliente
    // digita os dois direto no portal, sem precisar do link). So' compara digitos (ignora
    // pontos/traco/barra de qualquer lado) e a senha gravada em "Senha Cliente". Nunca revela
    // qual dos dois campos errou - so' "CPF/CNPJ ou senha invalidos", pra nao ajudar tentativa
    // de adivinhar. Devolve os MESMOS campos curados do portal_processo (nunca CPF, endereco,
    // honorarios ou anotacoes internas), um item por processo desse cliente.
    if (acao === 'portal_login') {
      try {
        const soDigitos = function(v) { return String(v || '').replace(/\D/g, ''); };
        // Completa com zero(s) a esquerda ate' 11 digitos (CPF) ou 14 (CNPJ): a celula da
        // planilha pode ter perdido um zero a esquerda do CPF se em algum momento foi salva
        // como numero em vez de texto (comum em registros antigos, de antes da mascara de
        // CPF/CNPJ existir no formulario) - sem isso, um CPF que comeca com "0" nunca bateria.
        const padDoc_ = function(d) { if (!d) return d; return d.length <= 11 ? d.padStart(11, '0') : d.padStart(14, '0'); };
        const cpfBuscado = padDoc_(soDigitos(body.cpfCnpj));
        const senhaBuscada = String(body.senha || '').trim();
        if (!cpfBuscado || !senhaBuscada) return json_({ ok: false, error: 'Informe CPF/CNPJ e senha.' });
        const sheetLogin = ss.getSheetByName(ABA_PROCESSOS);
        if (!sheetLogin) return json_({ ok: false, error: 'CPF/CNPJ ou senha invalidos.' });
        const dataLogin = sheetLogin.getDataRange().getValues();
        if (dataLogin.length < 2) return json_({ ok: false, error: 'CPF/CNPJ ou senha invalidos.' });
        const headersLogin = dataLogin[0].map(function(h) { return String(h).trim(); });
        const idxCpf = headersLogin.indexOf('CPF/CNPJ');
        const idxSenha = headersLogin.indexOf('Senha Cliente');
        if (idxCpf < 0 || idxSenha < 0) return json_({ ok: false, error: 'CPF/CNPJ ou senha invalidos.' });
        const MAPA_PORTAL_LOGIN = {
          'Nome': 'nome', 'Número do Processo': 'processo', 'Ramo do Direito': 'ramo',
          'Tipo de Ação': 'tipo', 'Status': 'status', 'proximoPassoAdv': 'proximoPasso',
          'Providências para o Cliente Atender': 'providenciaCliente', 'dataPrazoCliente': 'dataPrazoCliente',
          'Data Audiência': 'dataAudiencia', 'Horário Audiência': 'horaAudiencia',
          'Local da Audiência': 'localAudiencia', 'Última Atualização': 'ultimaAtualizacao',
          'Data Atualização': 'dataAtualizacao', 'Parte': 'parte', 'Outra Parte': 'outra'
        };
        const idxCamposLogin = {};
        Object.keys(MAPA_PORTAL_LOGIN).forEach(function(nomeCol) { idxCamposLogin[nomeCol] = headersLogin.indexOf(nomeCol); });
        const processos = [];
        for (let i = 1; i < dataLogin.length; i++) {
          const linhaAtual = dataLogin[i];
          if (padDoc_(soDigitos(linhaAtual[idxCpf])) !== cpfBuscado) continue;
          if (String(linhaAtual[idxSenha] || '').trim() !== senhaBuscada) continue;
          const obj = {};
          Object.keys(MAPA_PORTAL_LOGIN).forEach(function(nomeCol) {
            const idx = idxCamposLogin[nomeCol];
            let v = idx >= 0 ? linhaAtual[idx] : '';
            if (v instanceof Date) v = Utilities.formatDate(v, Session.getScriptTimeZone(), 'dd/MM/yyyy');
            obj[MAPA_PORTAL_LOGIN[nomeCol]] = v !== null && v !== undefined ? String(v) : '';
          });
          processos.push(obj);
        }
        if (!processos.length) return json_({ ok: false, error: 'CPF/CNPJ ou senha invalidos.' });
        return json_({ ok: true, processos: processos });
      } catch (errLogin) {
        return json_({ ok: false, error: 'Nao foi possivel entrar agora. Tente novamente.' });
      }
    }

    if (acao === 'excluir_processo') {
      const sheet = ss.getSheetByName(ABA_PROCESSOS);
      if (!sheet) return json_({ ok: false, error: 'Aba nao encontrada' });
      const linha = parseInt(body.linha);
      if (!linha || linha < 2) return json_({ ok: false, error: 'Linha invalida' });
      sheet.deleteRow(linha);
      SpreadsheetApp.flush();
      return json_({ ok: true, acao: 'excluir_processo', linha: linha });
    }

    // HONORARIOS
    if (acao === 'lancar_honorario') {
      const sheet = garantirAbaHonorarios_();
      const campos = body.campos;
      const ultimaLinha = sheet.getLastRow();
      let proximoId = 1;
      if (ultimaLinha > 1) {
        const ids = sheet.getRange(2, 1, ultimaLinha - 1, 1).getValues()
          .flat().map(v => parseInt(v)).filter(v => !isNaN(v) && v > 0);
        if (ids.length > 0) proximoId = Math.max.apply(null, ids) + 1;
      }
      const novaLinha = HEADERS_HON.map(function(h, idx) {
        if (idx === 0) return proximoId;
        if (h === 'Data Lancamento') return Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'dd/MM/yyyy');
        return campos[h] || '';
      });
      sheet.appendRow(novaLinha);
      SpreadsheetApp.flush();
      return json_({ ok: true, id: proximoId });
    }

    if (acao === 'atualizar_honorario') {
      const sheet = garantirAbaHonorarios_();
      const data = sheet.getDataRange().getValues();
      const headers = data[0].map(h => String(h).trim());
      const id = parseInt(body.id);
      let linhaEncontrada = -1;
      for (let i = 1; i < data.length; i++) {
        if (parseInt(data[i][0]) === id) { linhaEncontrada = i + 1; break; }
      }
      if (linhaEncontrada < 0) return json_({ ok: false, error: 'Lancamento nao encontrado' });
      const campos = body.campos;
      Object.keys(campos).forEach(function(nomeCol) {
        const colIdx = headers.indexOf(nomeCol);
        if (colIdx >= 0) sheet.getRange(linhaEncontrada, colIdx + 1).setValue(campos[nomeCol]);
      });
      SpreadsheetApp.flush();
      return json_({ ok: true, id: id });
    }

    if (acao === 'excluir_honorario') {
      const sheet = garantirAbaHonorarios_();
      const data = sheet.getDataRange().getValues();
      const id = parseInt(body.id);
      let linhaEncontrada = -1;
      for (let i = 1; i < data.length; i++) {
        if (parseInt(data[i][0]) === id) { linhaEncontrada = i + 1; break; }
      }
      if (linhaEncontrada < 0) return json_({ ok: false, error: 'Lancamento nao encontrado' });
      sheet.deleteRow(linhaEncontrada);
      SpreadsheetApp.flush();
      return json_({ ok: true });
    }

    // Numero sequencial e definitivo do recibo, atribuido so na hora de imprimir/gerar PDF.
    // LockService garante que dois recibos impressos ao mesmo tempo (ex.: duas abas abertas)
    // nunca saiam com o mesmo numero.
    if (acao === 'gerar_numero_recibo') {
      const lockRec = LockService.getScriptLock();
      lockRec.waitLock(15000);
      try {
        const sheetRec = garantirAbaGen_(ABA_RECIBOS, HEADERS_RECIBOS);
        const tz = Session.getScriptTimeZone();
        const ano = parseInt(Utilities.formatDate(new Date(), tz, 'yyyy'));
        const ultimaLinhaRec = sheetRec.getLastRow();
        let seq = 1;
        if (ultimaLinhaRec > 1) {
          const anos = sheetRec.getRange(2, 2, ultimaLinhaRec - 1, 1).getValues().flat();
          const seqs = sheetRec.getRange(2, 3, ultimaLinhaRec - 1, 1).getValues().flat();
          let maxSeqAno = 0;
          for (let i = 0; i < anos.length; i++) {
            if (parseInt(anos[i]) === ano) {
              const s = parseInt(seqs[i]);
              if (!isNaN(s) && s > maxSeqAno) maxSeqAno = s;
            }
          }
          seq = maxSeqAno + 1;
        }
        const numero = ('0000' + seq).slice(-4) + '/' + ano;
        const campos = body.campos || {};
        sheetRec.appendRow([
          numero, ano, seq,
          Utilities.formatDate(new Date(), tz, 'dd/MM/yyyy HH:mm'),
          campos['Cliente'] || '', campos['CPF/CNPJ'] || '', campos['Processo'] || '',
          campos['Valor Total'] || '', campos['Forma Pagamento'] || ''
        ]);
        SpreadsheetApp.flush();
        return json_({ ok: true, numero: numero, ano: ano, sequencial: seq });
      } finally {
        lockRec.releaseLock();
      }
    }

        // CONTAS A PAGAR
    if (acao === 'lancar_conta') {
      const sheet = garantirAbaContas_();
      const campos = body.campos;
      const ultimaLinha = sheet.getLastRow();
      let proximoId = 1;
      if (ultimaLinha > 1) {
        const ids = sheet.getRange(2, 1, ultimaLinha - 1, 1).getValues()
          .flat().map(v => parseInt(v)).filter(v => !isNaN(v) && v > 0);
        if (ids.length > 0) proximoId = Math.max.apply(null, ids) + 1;
      }
      const novaLinha = HEADERS_CONTAS.map(function(h, idx) {
        if (idx === 0) return proximoId;
        if (h === 'Data Lancamento') return Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'dd/MM/yyyy');
        return campos[h] || '';
      });
      sheet.appendRow(novaLinha);
      SpreadsheetApp.flush();
      return json_({ ok: true, id: proximoId });
    }

    if (acao === 'atualizar_conta') {
      const sheet = garantirAbaContas_();
      const data = sheet.getDataRange().getValues();
      const id = parseInt(body.id);
      let linha = -1;
      for (let i = 1; i < data.length; i++) {
        if (parseInt(data[i][0]) === id) { linha = i + 1; break; }
      }
      if (linha < 0) return json_({ ok: false, error: 'Conta nao encontrada' });
      const campos = body.campos || {};
      HEADERS_CONTAS.forEach(function(h, idx) {
        if (idx === 0 || h === 'Data Lancamento') return;
        if (campos.hasOwnProperty(h)) sheet.getRange(linha, idx + 1).setValue(campos[h]);
      });
      SpreadsheetApp.flush();
      return json_({ ok: true });
    }

    if (acao === 'excluir_conta') {
      const sheet = garantirAbaContas_();
      const data = sheet.getDataRange().getValues();
      const id = parseInt(body.id);
      let linha = -1;
      for (let i = 1; i < data.length; i++) {
        if (parseInt(data[i][0]) === id) { linha = i + 1; break; }
      }
      if (linha < 0) return json_({ ok: false, error: 'Conta nao encontrada' });
      sheet.deleteRow(linha);
      SpreadsheetApp.flush();
      return json_({ ok: true });
    }
    if (acao === 'salvar_config') {
  const sheet = garantirAbaGen_(ABA_CONFIGP, HEADERS_CONFIGP);
  const data = sheet.getDataRange().getValues();
  const chave = String(body.chave || '');
  let linha = -1;
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === chave) { linha = i + 1; break; }
  }
  if (linha < 0) { sheet.appendRow([chave, body.valor || '']); }
  else { sheet.getRange(linha, 2).setValue(body.valor || ''); }
  SpreadsheetApp.flush();
  return json_({ ok: true });
}
    const rAtend = crudGen_(acao, 'atend', ABA_ATEND, HEADERS_ATEND, body);
if (rAtend) return rAtend;
    const rEntrega = crudGen_(acao, 'entrega', ABA_ENTREGAS, HEADERS_ENTREGAS, body);
    if (rEntrega) return rEntrega;
    const rAcc = crudGen_(acao, 'acc', ABA_ACC, HEADERS_ACC, body);
    if (rAcc) return rAcc;
    const rAcl = crudGen_(acao, 'acl', ABA_ACL, HEADERS_ACL, body);
    if (rAcl) return rAcl;
    const rLoccop = crudGen_(acao, 'loccop', ABA_LOCCOP, HEADERS_LOCCOP, body);
    if (rLoccop) return rLoccop;

    // COMISSOES IMOBILIARIAS
    const rCorretor = crudGen_(acao, 'corretor', ABA_CORRETORES, HEADERS_CORRETORES, body);
    if (rCorretor) return rCorretor;
    const rCom = crudGen_(acao, 'com', ABA_COM, HEADERS_COM, body);
    if (rCom) return rCom;
    const rProposta = crudGen_(acao, 'proposta', ABA_PROPOSTAS, HEADERS_PROPOSTAS, body);
    if (rProposta) return rProposta;
    const rComRateio = crudGen_(acao, 'comRateio', ABA_COM_RATEIO, HEADERS_COM_RATEIO, body);
    if (rComRateio) return rComRateio;
    const rComReciboReceb = crudGen_(acao, 'comReciboReceb', ABA_COM_RECIBOS_RECEB, HEADERS_COM_RECIBOS_RECEB, body);
    if (rComReciboReceb) return rComReciboReceb;
    const rComReciboRepasse = crudGen_(acao, 'comReciboRepasse', ABA_COM_RECIBOS_REPASSE, HEADERS_COM_RECIBOS_REPASSE, body);
    if (rComReciboRepasse) return rComReciboRepasse;

    // IMÓVEIS DISPONÍVEIS (cadastro interno da equipe — CRUD genérico, mesmo padrão acima)
    const rImov = crudGen_(acao, 'imov', ABA_IMOV, HEADERS_IMOV, body);
    if (rImov) return rImov;

    const rImovAt = crudGen_(acao, 'imovAt', ABA_IMOV_AT, HEADERS_IMOV_AT, body);
    if (rImovAt) return rImovAt;

    const rNotas = crudGen_(acao, 'notas', ABA_NOTAS, HEADERS_NOTAS, body);
    if (rNotas) return rNotas;

    // TAREFAS (rascunhos) - sincroniza com o Google Agenda (Eixo -> Agenda, mao unica) sempre
    // que uma tarefa e criada/atualizada/excluida. Ver tarefaSincronizarCalendar_ e nota no
    // topo do arquivo (HEADERS_RASCUNHOS). Falha de calendario nunca deve derrubar o
    // salvamento da tarefa em si - por isso o try/catch envolvendo so a parte de calendario.
    if (acao === 'excluir_rascunhos') {
      try { tarefaRemoverEventoCalendar_(parseInt(body.id)); } catch (errCalDel) {}
    }
    const rRascunhos = crudGen_(acao, 'rascunhos', ABA_RASCUNHOS, HEADERS_RASCUNHOS, body);
    if (rRascunhos) {
      if (acao === 'lancar_rascunhos' || acao === 'atualizar_rascunhos') {
        try {
          const rRascResp = JSON.parse(rRascunhos.getContent());
          if (rRascResp && rRascResp.ok) {
            const idTarefaSync = (acao === 'lancar_rascunhos') ? rRascResp.id : parseInt(body.id);
            tarefaSincronizarCalendar_(idTarefaSync);
          }
        } catch (errCalSync) {}
      }
      return rRascunhos;
    }

    const rPartes = crudGen_(acao, 'partes', ABA_PARTES, HEADERS_PARTES, body);
    if (rPartes) return rPartes;

    const rColoc = crudGen_(acao, 'coloc', ABA_COLOC, HEADERS_COLOC, body);
    if (rColoc) return rColoc;

    const rLocLoc = crudGen_(acao, 'locloc', ABA_LOCLOC, HEADERS_LOCLOC, body);
    if (rLocLoc) return rLocLoc;

    const rAudit = crudGen_(acao, 'audit', ABA_AUDIT, HEADERS_AUDIT, body);
    if (rAudit) return rAudit;

        // LOCACOES
    if (acao === 'lancar_locc' || acao === 'atualizar_locc') {
      // Trava de seguranca contra contrato "vazio" (sem locador/locatario/endereco) indo
      // parar na planilha - foi assim que um contrato fantasma apareceu e acabou sendo
      // apagado por engano. O formulario ja valida isso, mas qualquer chamada direta
      // (fora do formulario) tambem precisa respeitar a regra.
      const camposChk = body.campos || {};
      const faltando = ['Contrato', 'Locador', 'Locatario', 'Endereco'].filter(function (k) {
        return !String(camposChk[k] || '').trim();
      });
      if (faltando.length) {
        return json_({ ok: false, error: 'Contrato nao gravado: preencha ' + faltando.join(', ') + '.' });
      }
    }
    const rLocc = crudGen_(acao, 'locc', ABA_LOCC, HEADERS_LOCC, body);
    if (rLocc) return rLocc;
    const rLocl = crudGen_(acao, 'locl', ABA_LOCL, HEADERS_LOCL, body);
    if (rLocl) return rLocl;

    const rLocreaj = crudGen_(acao, 'locreaj', ABA_LOCREAJ, HEADERS_LOCREAJ, body);
    if (rLocreaj) return rLocreaj;

    const rSin = crudGen_(acao, 'sinistro', ABA_SIN, HEADERS_SIN, body);
    if (rSin) return rSin;

    const rSinHist = crudGen_(acao, 'sinistroHist', ABA_SINHIST, HEADERS_SINHIST, body);
    if (rSinHist) return rSinHist;

    if (acao === 'gerar_mes_loc') {
      const mes = String(body.mes || '');
      if (!/^[0-9]{2}\/[0-9]{4}$/.test(mes)) return json_({ ok: false, error: 'Mes invalido, use MM/AAAA' });
      const sc = garantirAbaGen_(ABA_LOCC, HEADERS_LOCC);
      const sl = garantirAbaGen_(ABA_LOCL, HEADERS_LOCL);
      const dc = sc.getDataRange().getValues();
      const dl = sl.getDataRange().getValues();
      // Lê a aba de lançamentos por NOME de coluna (não por índice fixo) — HEADERS_LOCL cresceu
      // com o tempo (IPTU Parcela Numero/Total, Recibo Enviado...) e ler por posição quebraria
      // se a ordem das colunas na planilha não bater exatamente com o array acima.
      const hl = dl.length ? dl[0].map(function(h){ return String(h).trim(); }) : HEADERS_LOCL;
      const idxL = {}; hl.forEach(function(h, i){ idxL[h] = i; });

      const existentes = {};
      for (let i = 1; i < dl.length; i++) existentes[String(getL(dl[i], 'Mes Ref')) + '|' + String(getL(dl[i], 'Contrato'))] = true;

      // Acha, pra cada contrato, o lançamento mais recente já existente — é dele que o mês novo
      // herda o ALUGUEL e os ENCARGOS (histórico realmente lançado, não o cadastro do contrato,
      // que pode estar desatualizado — pedido do usuário) e as cobranças recorrentes/parceladas.
      const ultimoPorContrato = {};
      for (let i = 1; i < dl.length; i++) {
        const row = dl[i];
        const contratoLin = String(getL(row, 'Contrato'));
        const mesLin = String(getL(row, 'Mes Ref'));
        const partes = mesLin.split('/');
        if (partes.length !== 2) continue;
        const chaveOrdem = partes[1] + partes[0]; // AAAAMM, pra comparar cronologicamente
        if (!ultimoPorContrato[contratoLin] || chaveOrdem > ultimoPorContrato[contratoLin].chaveOrdem) {
          ultimoPorContrato[contratoLin] = {
            chaveOrdem: chaveOrdem,
            aluguel: numBr_(getL(row, 'Aluguel')),
            condominio: numBr_(getL(row, 'Condominio')),
            aguaPriv: numBr_(getL(row, 'Agua Privativa')),
            gasPriv: numBr_(getL(row, 'Gas Privativa')),
            aguaComum: numBr_(getL(row, 'Agua Comum')),
            gasPiscina: numBr_(getL(row, 'Gas Piscina')),
            energiaComum: numBr_(getL(row, 'Energia Comum')),
            taxaLeitura: numBr_(getL(row, 'Taxa Leitura')),
            fundoFerias: numBr_(getL(row, 'Fundo Ferias')),
            decimoTerceiro: numBr_(getL(row, 'Decimo Terceiro')),
            desconto: numBr_(getL(row, 'Desconto')),
            descontoIncideTaxa: String(getL(row, 'Desconto Incide Taxa') || '').trim(),

            iptuParcelaNum: parseInt(getL(row, 'IPTU Parcela Numero'), 10) || 0,

      let id = proximoId_(sl);
      let criados = 0;
      for (let i = 1; i < dc.length; i++) {
        const r = dc[i];
        if (!r[1]) continue;
        if (String(r[9]).toLowerCase().indexOf('sim') !== 0) continue;
        const contratoNr = String(r[1]);
        if (existentes[mes + '|' + contratoNr]) continue;
        const ultimo = ultimoPorContrato[contratoNr];

        // Aluguel e condomínio: preferem o valor REALMENTE LANÇADO no mês mais recente; só caem

        // nas "Outras Cobranças" parceladas (continuarOutrasCobrancas_). Sem contagem de parcela,
        // o IPTU é tratado como valor fixo recorrente (repete igual ao mês anterior, como o

        var iptu = 0, iptuParcelaNum = '', iptuParcelaTotal = '';

            iptu = ultimo.iptu;
            iptuParcelaNum = ultimo.iptuParcelaNum + 1;
            iptuParcelaTotal = ultimo.iptuParcelaTotal;
          } // já chegou na última parcela — não repete mais (IPTU sai zerado este mês)
        } else if (ultimo) {
          iptu = ultimo.iptu;
        } else {
          iptu = numBr_(r[12]);
        }

        var outrasDescNova = continuarOutrasCobrancas_(ultimo ? ultimo.outrasDesc : '');
        // Saldo pendente de "recebido a menor/maior" do mês anterior, quando o usuário escolheu

        // — só este mês novo recebe o item, não continua se um 3º mês for gerado a partir dele).
        if (ultimo && ultimo.saldoProxMesValor) {
          var descSaldo = (ultimo.saldoProxMesDesc || 'Saldo do mês anterior').replace(/::/g, '-').replace(/\|\|/g, '-');
          var itemSaldo = descSaldo + '::' + ultimo.saldoProxMesValor + '::pontual::0::0::repasse::0';
          outrasDescNova = outrasDescNova ? (outrasDescNova + '||' + itemSaldo) : itemSaldo;
        }

        // Observações (do locador/locatário) sempre vazias num mês recém-gerado — não faz

        const campos = {
          'Mes Ref': mes, 'Contrato': contratoNr, 'Locador': r[2], 'Locatario': r[3], 'Endereco': r[4],
          'Aluguel': aluguel, 'Taxa Adm': taxa.toFixed(2).replace('.', ','), 'Valor Repasse': '',
          'Dia Vencimento': r[7], 'Dia Repasse': r[8], 'Status': 'Pendente', 'Data Pagamento': '',
          'Origem Pgto': '', 'Data Repasse': '', 'Status Repasse': 'Pendente',
          'Condominio': condominio, 'IPTU': iptu, 'IPTU Parcela Numero': iptuParcelaNum, 'IPTU Parcela Total': iptuParcelaTotal,
          'Outras Cobrancas Desc': outrasDescNova, 'Outras Cobrancas Valor': 0,
          'Agua Privativa': ultimo ? ultimo.aguaPriv : 0, 'Gas Privativa': ultimo ? ultimo.gasPriv : 0,
          'Agua Comum': ultimo ? ultimo.aguaComum : 0, 'Gas Piscina': ultimo ? ultimo.gasPiscina : 0,
          'Energia Comum': ultimo ? ultimo.energiaComum : 0, 'Taxa Leitura': ultimo ? ultimo.taxaLeitura : 0,
          'Fundo Ferias': ultimo ? ultimo.fundoFerias : 0, 'Decimo Terceiro': ultimo ? ultimo.decimoTerceiro : 0,
          'Desconto': ultimo ? ultimo.desconto : 0, 'Multa': 0, 'Juros': 0,
          'Desconto Incide Taxa': ultimo && ultimo.descontoIncideTaxa ? ultimo.descontoIncideTaxa : 'Nao',
          'Observacoes': '', 'Observacoes Locatario': '',
          'Recibo Enviado': 'Nao', 'Data Envio Recibo': '',
          'Valor Recebido': '', 'Saldo Prox Mes Desc': '', 'Saldo Prox Mes Valor': ''
        };
        const novaLinha = HEADERS_LOCL.map(function(h, idx) {
          if (idx === 0) return id;
          return Object.prototype.hasOwnProperty.call(campos, h) ? campos[h] : '';
        });
        sl.appendRow(novaLinha);
        id++; criados++;
      }
      SpreadsheetApp.flush();
      return json_({ ok: true, criados: criados });
    }

  if (acao === 'criar_pasta_processo') {
  var raizP = getOuCriarPastaRaiz_();
  var nomeProc = String(body.nome || 'Processo sem nome').trim();
  // Arquiva na subpasta da letra inicial do nome (A, B, C...), igual a organizacao que o
  // usuario ja usa manualmente pra clientes - configurado em 02/set/2026.
  var subpastaLetra = getOuCriarSubpastaLetra_(raizP, letraInicialCliente_(nomeProc));
  var pastaNova = subpastaLetra.createFolder(nomeProc);
  pastaNova.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return json_({ ok: true, id: pastaNova.getId(), url: pastaNova.getUrl() });
  }
  // Migracao unica (02/set/2026): move as pastas de cliente criadas ANTES da mudanca de
  // 'criar_pasta_processo' (soltas na antiga raiz 'Eixo - Documentos'/'LexFlow - Documentos'
  // na raiz do Drive) pra dentro da subpasta de letra certa, na nova pasta raiz configurada em
  // ID_PASTA_RAIZ_CLIENTES. So move pastas (arquivos soltos na raiz antiga, se houver, ficam
  // onde estao). Idempotente: rodar de novo depois que a raiz antiga ja esvaziou nao faz nada.
  if (acao === 'migrar_pastas_clientes_letra') {
  var nomesRaizAntiga = ['Eixo - Documentos', 'LexFlow - Documentos'];
  var raizAntiga = null;
  for (var iRA = 0; iRA < nomesRaizAntiga.length; iRA++) {
    var itRA = DriveApp.getRootFolder().getFoldersByName(nomesRaizAntiga[iRA]);
    if (itRA.hasNext()) { raizAntiga = itRA.next(); break; }
  }
  if (!raizAntiga) return json_({ ok: false, error: 'Pasta antiga (Eixo - Documentos) nao encontrada na raiz do Drive - nada pra migrar.' });
  var raizNova = getOuCriarPastaRaiz_();
  if (raizAntiga.getId() === raizNova.getId()) return json_({ ok: false, error: 'Raiz antiga e nova sao a mesma pasta - nada pra migrar.' });
  var detalhes = [];

  while (itFolders.hasNext()) {
    var fOld = itFolders.next();
    var nomeCli = fOld.getName();
    var letra = letraInicialCliente_(nomeCli);

  }
  return json_({ ok: true, movidas: detalhes.length, detalhes: detalhes });
  }
  if (acao === 'listar_arquivos_pasta') {
  var idPasta = extrairIdPasta_(body.pasta);
  if (!idPasta) return json_({ ok: false, error: 'Pasta nao informada' });
  var pastaObj;
  try { pastaObj = DriveApp.getFolderById(idPasta); } catch (eDrive) { return json_({ ok: false, error: 'Pasta nao encontrada ou sem acesso' }); }
  var listaArq = [];
  var itArq = pastaObj.getFiles();
  while (itArq.hasNext()) {
  var fArq = itArq.next();
  listaArq.push({ id: fArq.getId(), nome: fArq.getName(), url: fArq.getUrl(), tipo: fArq.getMimeType(), tamanho: fArq.getSize(), atualizado: Utilities.formatDate(fArq.getLastUpdated(), Session.getScriptTimeZone() || 'America/Sao_Paulo', 'dd/MM/yyyy HH:mm') });

  if (acao === 'upload_arquivo') {

  var bytesArq = Utilities.base64Decode(body.base64);

  // ENVIAR E-MAIL A PARTIR DE UMA TAREFA (30/ago/2026) - reaproveita MailApp (mesmo servico ja

        SpreadsheetApp.flush();
      }
    } catch (errLogEmail) {}

  }

  // ESCANEAR GMAIL POR TAREFAS NOVAS (30/ago/2026) - execucao manual via doPost (alem do
  // trigger periodico configurado por lfConfigurarTriggerGmailTarefas_). Ver
  // gmailCriarTarefasDeEmail_ para o funcionamento completo.
  if (acao === 'gmail_scan_tarefas') {
    try { return json_(gmailCriarTarefasDeEmail_()); }
    catch (errGmailScan) { return json_({ ok: false, error: String(errGmailScan) }); }
  }

    return json_({ ok: false, error: 'Acao desconhecida: ' + acao });

  } catch (err) {
    return json_({ ok: false, error: err.message });
  }
}





function migrarStatusRepasseLocl_() {
  const sheet = garantirAbaGen_(ABA_LOCL, HEADERS_LOCL);
  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return 'Nenhum lancamento.';
  const headers = data[0].map(function(h){ return String(h).trim(); });
  const idxStatus = headers.indexOf('Status');
  const idxStatusRep = headers.indexOf('Status Repasse');
  if (idxStatus < 0 || idxStatusRep < 0) return 'Colunas nao encontradas: ' + idxStatus + '/' + idxStatusRep;
  let migrados = 0;
  for (let i = 1; i < data.length; i++) {
    if (!data[i][0]) continue;
    const atual = String(data[i][idxStatusRep] || '').trim();
    if (atual) continue;
    const statusAntigo = String(data[i][idxStatus] || '').trim();
    const novoValor = (statusAntigo === 'Pago') ? 'Repassado' : 'Pendente';
    sheet.getRange(i + 1, idxStatusRep + 1).setValue(novoValor);
    migrados++;
  }
  SpreadsheetApp.flush();
  return 'Migrados: ' + migrados;
}

// TAREFAS <-> GOOGLE AGENDA (30/ago/2026) - sincronizacao MAO UNICA (Eixo -> Agenda).
// Editar/mover um evento direto no Google Agenda NAO volta pro Eixo - a Tarefa continua
// sendo a fonte de verdade. Usa CalendarApp.getDefaultCalendar() (a mesma conta
// "adeborges@gmail.com" que ja aparece na lista TODAS_AGENDAS de obterAgenda_ acima), entao
// os eventos criados aqui tambem aparecem sozinhos no widget "Agenda do dia" da Home (como
// evento de dia inteiro -> vira item da lista "tarefas" daquele widget, ver agendaDoDia).
// Só cria/mantém evento quando ha Data Follow-up preenchida E a tarefa ainda esta
// "Em andamento"; quando o follow-up é apagado ou o status muda (Concluída/Convertido/
// Descartado), o evento correspondente é removido da Agenda automaticamente.
function tarefaSincronizarCalendar_(id) {
  if (!id) return;
  var sheet = garantirAbaGen_(ABA_RASCUNHOS, HEADERS_RASCUNHOS);
  var data = sheet.getDataRange().getValues();
  var idxData = HEADERS_RASCUNHOS.indexOf('Data Follow-up');
  var idxStatus = HEADERS_RASCUNHOS.indexOf('Status');
  var idxTitulo = HEADERS_RASCUNHOS.indexOf('Titulo');
  var idxTipo = HEADERS_RASCUNHOS.indexOf('Tipo');
  var idxNome = HEADERS_RASCUNHOS.indexOf('Nome');
  var idxEvt = HEADERS_RASCUNHOS.indexOf('Calendar Event ID');
  var idxProc = HEADERS_RASCUNHOS.indexOf('Nr Processo');
  var idxSin = HEADERS_RASCUNHOS.indexOf('Sinistro ID');
  var linha = -1;
  for (var i = 1; i < data.length; i++) {
    if (parseInt(data[i][0]) === id) { linha = i + 1; break; }
  }
  if (linha < 0) return;
  var row = data[linha - 1];
  var dataFollow = row[idxData];
  var status = String(row[idxStatus] || '').trim();
  var eventoIdAtual = String(row[idxEvt] || '').trim();
  var cal = CalendarApp.getDefaultCalendar();

  var deveTerEvento = !!dataFollow && (status === 'Em andamento' || !status);
  if (!deveTerEvento) {
    if (eventoIdAtual) {
      try { var evOld = cal.getEventById(eventoIdAtual); if (evOld) evOld.deleteEvent(); } catch (eDelCal) {}
      sheet.getRange(linha, idxEvt + 1).setValue('');
    }
    return;
  }

  var dataEvento = (dataFollow instanceof Date) ? dataFollow : rascParseDataParaDate_(dataFollow);
  if (!dataEvento || isNaN(dataEvento.getTime())) return;

  var nrProcesso = idxProc >= 0 ? String(row[idxProc] || '').trim() : '';
  var sinistroId = idxSin >= 0 ? String(row[idxSin] || '').trim() : '';
  var titulo = '🗒️ ' + String(row[idxTitulo] || row[idxNome] || row[idxTipo] || 'Tarefa') + (nrProcesso ? ' (Proc. ' + nrProcesso + ')' : '') + (sinistroId ? ' (Sinistro)' : '') + ' — Eixo';
  var descricao = 'Tarefa "' + String(row[idxTipo] || '') + '"' + (row[idxNome] ? ' — ' + String(row[idxNome]) : '')
    + (nrProcesso ? '\nProcesso vinculado: ' + nrProcesso : '')
    + (sinistroId ? '\nSinistro de garantia locatícia vinculado (ID ' + sinistroId + ')' : '')
    + '\n\nCriado automaticamente pelo sistema Eixo a partir do Follow-up da tarefa.'
    + ' Editar a data aqui na Agenda NAO atualiza a tarefa no painel - mude o Follow-up direto no Eixo.';

  if (eventoIdAtual) {
    try {
      var evExist = cal.getEventById(eventoIdAtual);
      if (evExist) {
        evExist.setTitle(titulo);
        evExist.setDescription(descricao);
        var inicioAtual = evExist.getStartTime();
        if (Utilities.formatDate(inicioAtual, Session.getScriptTimeZone() || 'America/Sao_Paulo', 'yyyy-MM-dd')
          !== Utilities.formatDate(dataEvento, Session.getScriptTimeZone() || 'America/Sao_Paulo', 'yyyy-MM-dd')) {
          evExist.setAllDayDate(dataEvento);
        }
        return;
      }
    } catch (eUpdCal) {}
  }
  var novoEvento = cal.createAllDayEvent(titulo, dataEvento, { description: descricao });
  sheet.getRange(linha, idxEvt + 1).setValue(novoEvento.getId());
}

// Chamada ANTES de excluir a linha da tarefa (crudGen_ apaga a linha inteira, entao o
// Calendar Event ID precisa ser lido e removido antes disso, senão fica orfao na Agenda).
function tarefaRemoverEventoCalendar_(id) {
  if (!id) return;
  var sheet = garantirAbaGen_(ABA_RASCUNHOS, HEADERS_RASCUNHOS);
  var data = sheet.getDataRange().getValues();
  var idxEvt = HEADERS_RASCUNHOS.indexOf('Calendar Event ID');
  for (var i = 1; i < data.length; i++) {
    if (parseInt(data[i][0]) === id) {
      var eventoId = String(data[i][idxEvt] || '').trim();
      if (eventoId) {
        try {
          var cal = CalendarApp.getDefaultCalendar();
          var ev = cal.getEventById(eventoId);
          if (ev) ev.deleteEvent();
        } catch (eDelCal2) {}
      }
      break;
    }
  }
}

// 'Data Follow-up' é gravada como texto "dd/MM/yyyy" (mesmo padrão usado em todo o resto do
// sistema) - converte pra Date real, sem depender de new Date(string) (formato ambíguo entre
// locales). Retorna null se não for parseável.
function rascParseDataParaDate_(s) {
  var partes = String(s || '').trim().split('/');
  if (partes.length !== 3) return null;
  var d = parseInt(partes[0], 10), m = parseInt(partes[1], 10), a = parseInt(partes[2], 10);
  if (!d || !m || !a) return null;
  return new Date(a, m - 1, d);
}

// TAREFAS <-> GMAIL (30/ago/2026): cria uma Tarefa automaticamente a partir de um e-mail
// marcado com o rótulo "Eixo/Nova Tarefa" no Gmail. Fluxo: usuário aplica o rótulo a um
// e-mail (do cliente, de um cartório, etc.) -> na próxima varredura (manual via
// acao=gmail_scan_tarefas, ou automática pelo gatilho configurado em
// lfConfigurarTriggerGmailTarefas_), o e-mail vira uma Tarefa (Tipo "Geral", pra você
// reclassificar manualmente no módulo certo) com o assunto, remetente e um trecho do corpo
// nas Notas, e o rótulo é trocado por "Eixo/Tarefa Criada" (não reprocessa a mesma thread).
// ATENÇÃO: isto usa GmailApp (ler/rotular e-mails), um escopo mais amplo que o MailApp já
// usado no resto do sistema (que só ENVIA, nunca lê a caixa). Na primeira execução (manual,
// pelo editor do Apps Script, rodando gmailCriarTarefasDeEmail_ ou
// lfConfigurarTriggerGmailTarefas_) o Google vai pedir uma autorização nova - isso só pode
// ser aceito pelo próprio dono da conta (Executar -> Revisar permissões -> Avançado ->
// acessar [nome do projeto] (não seguro) -> Permitir), nunca por automação.
var GMAIL_LABEL_TAREFA_NOVA = 'Eixo/Nova Tarefa';
var GMAIL_LABEL_TAREFA_CRIADA = 'Eixo/Tarefa Criada';

function gmailCriarTarefasDeEmail_() {
  var labelNova = GmailApp.getUserLabelByName(GMAIL_LABEL_TAREFA_NOVA) || GmailApp.createLabel(GMAIL_LABEL_TAREFA_NOVA);
  var labelCriada = GmailApp.getUserLabelByName(GMAIL_LABEL_TAREFA_CRIADA) || GmailApp.createLabel(GMAIL_LABEL_TAREFA_CRIADA);
  var threads = labelNova.getThreads(0, 30); // no maximo 30 por varredura, sobra pra proxima
  var sheet = garantirAbaGen_(ABA_RASCUNHOS, HEADERS_RASCUNHOS);
  var tz = Session.getScriptTimeZone() || 'America/Sao_Paulo';
  var criadas = 0;
  var erros = [];

  threads.forEach(function(thread) {
    try {
      var msg = thread.getMessages()[thread.getMessages().length - 1]; // a mais recente da thread
      var deNome = '';
      var deEmail = '';
      var remetente = String(msg.getFrom() || '');
      var matchEmail = remetente.match(/<([^>]+)>/);
      if (matchEmail) {
        deEmail = matchEmail[1];
        deNome = remetente.replace(/<[^>]+>/, '').replace(/["']/g, '').trim();
      } else {
        deEmail = remetente.trim();
      }
      var assunto = String(msg.getSubject() || '(sem assunto)');
      var corpo = String(msg.getPlainBody() || '').trim().slice(0, 800);
      var linkThread = 'https://mail.google.com/mail/u/0/#all/' + thread.getId();

      var id = proximoId_(sheet);
      var agora = Utilities.formatDate(new Date(), tz, 'dd/MM/yyyy');
      var novaLinha = HEADERS_RASCUNHOS.map(function(h) {
        if (h === 'ID') return id;
        if (h === 'Tipo') return 'Geral';
        if (h === 'Titulo') return assunto;
        if (h === 'Nome') return deNome || deEmail;
        if (h === 'Email') return deEmail;
        if (h === 'Notas') return '📧 Criada automaticamente a partir de um e-mail do Gmail.\nRemetente: ' + remetente
          + '\nLink da conversa: ' + linkThread + '\n\n' + corpo;
        if (h === 'Status') return 'Em andamento';
        if (h === 'Data Criacao') return agora;
        if (h === 'Data Atualizacao') return agora;
        return '';
      });
      sheet.appendRow(novaLinha);
      criadas++;

      thread.removeLabel(labelNova);
      thread.addLabel(labelCriada);
    } catch (errThread) {
      erros.push(String(errThread));
    }
  });
  if (criadas > 0) SpreadsheetApp.flush();
  return { ok: true, criadas: criadas, restantes: threads.length >= 30, erros: erros };
}

function lfConfigurarTriggerGmailTarefas_() {
  ScriptApp.getProjectTriggers().forEach(function(t) {
    if (t.getHandlerFunction() === 'gmailCriarTarefasDeEmail_') ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger('gmailCriarTarefasDeEmail_').timeBased().everyMinutes(15).create();
  return { ok: true, mensagem: 'Trigger de varredura do Gmail configurado (a cada 15 min).' };
}


function obterAgenda_() {

  var hoje = new Date();
  hoje.setHours(0, 0, 0, 0);

  function ehFimDeSemana(d) { var w = d.getDay(); return w === 0 || w === 6; }
  function addDias(d, n) { var r = new Date(d); r.setDate(r.getDate() + n); return r; }
  function proximoDiaUtil(d) { var r = new Date(d); while (ehFimDeSemana(r)) { r = addDias(r, 1); } return r; }

  var dia1 = proximoDiaUtil(hoje);
  var dia2 = proximoDiaUtil(addDias(dia1, 1));

  var TODAS_AGENDAS = [
    'adeborges@gmail.com',
    'd01425dc398f96028eed255ea4692026e53c7dd54e3c8aaf584f3d23a4d50945@group.calendar.google.com',
    'a0954ea61661971597e37bf067ae7e0c2ff9bdbfca12ccbf0b8d944e6e933add@group.calendar.google.com',
    '5d28b37f4a44f61267974ca184c9bf0f1fc00ed05627eb9bf31200db18972cd5@group.calendar.google.com'
  ];

    TODAS_AGENDAS.forEach(function(id) {

      try { evs = cal.getEventsForDay(dia); } catch (e2) { evs = []; }

        try { diaInteiro = ev.isAllDayEvent(); } catch (e3) { diaInteiro = false; }

        var item = {
          titulo: ev.getTitle(),
          hora: diaInteiro ? '' : Utilities.formatDate(inicio, tz, 'HH:mm'),
          local: ev.getLocation() || '',
          _ord: diaInteiro ? -1 : inicio.getTime()
        };
        if (diaInteiro) { tarefas.push(item); } else { compromissos.push(item); }
      });
    });
    compromissos.sort(function(a, b) { return a._ord - b._ord; });

    return capitaliza(NOMES_DIA[d.getDay()]) + ', ' + Utilities.formatDate(d, tz, 'dd/MM');
  }

  var ag1 = agendaDoDia(dia1);
  var ag2 = agendaDoDia(dia2);

  return {

        data: Utilities.formatDate(dia1, tz, 'dd/MM/yyyy'),

        label: labelDia(dia2),
        compromissos: ag2.compromissos,
        tarefas: ag2.tarefas
      }
    ]
  };

function sheetParaObjetos_(sheet) {
  if (!sheet) return [];
  var rows = sheet.getDataRange().getValues();
  if (rows.length < 2) return [];
  var headers = rows[0].map(function(h){ return String(h || '').trim(); });
  var out = [];
  for (var i = 1; i < rows.length; i++) {
    var row = rows[i];
    var obj = {};
    var vazio = true;
    headers.forEach(function(h, j) {
      if (!h) return;
      var v = row[j];
      obj[h] = v;
      if (v !== '' && v !== null && v !== undefined) vazio = false;
    });
    if (!vazio) out.push(obj);
  }
  return out;
}

function lfNbr_(v) {
  v = String(v == null ? '' : v).trim();
  if (!v) return 0;
  if (v.indexOf(',') >= 0) v = v.replace(/[.]/g, '').replace(',', '.');
  var n = parseFloat(v);
  return isNaN(n) ? 0 : n;
}

function lfParaData_(v) {
  if (v instanceof Date) return isNaN(v) ? null : v;
  var s = String(v == null ? '' : v).trim();
  if (!s) return null;
  var p = s.split(/[/-]/);
  if (p.length === 3) {
    if (p[0].length === 4) return new Date(+p[0], +p[1] - 1, +p[2]);
    var d = new Date(+p[2], +p[1] - 1, +p[0]);
    return isNaN(d) ? null : d;
  }
  var d2 = new Date(s);
  return isNaN(d2) ? null : d2;
}

function lfDiaOnly_(v) {
  var m = String(v == null ? '' : v).trim().match(/^([0-9]{1,2})/);
  return m ? parseInt(m[1], 10) : 0;
}

function lfClampDay_(y, m, d) {
  var last = new Date(y, m + 1, 0).getDate();
  return new Date(y, m, Math.min(Math.max(d, 1), last));
}

function lfPad2_(n) { return ('0' + n).slice(-2); }

function lfRefKey_(s) {
  s = String(s || '').toLowerCase().trim();
  if (!s) return null;
  if (s.indexOf('entr') === 0) return 'E';
  if (s.indexOf('contrato') === 0) return 'C';
  var m = s.match(/^([0-9]{1,2})/);
  return m ? parseInt(m[1], 10) : null;
}

function lfDiasAte_(data, hoje) {
  if (!data) return null;
  var d = new Date(data);
  d.setHours(0, 0, 0, 0);
  return Math.round((d - hoje) / 86400000);
}

function lfLinhaPrazo_(dias) {
  if (dias < 0) return 'Venceu ha ' + Math.abs(dias) + ' dia(s)';
  if (dias === 0) return 'Vence hoje';
  return 'Faltam ' + dias + ' dia(s)';
}

function lfColetarProcessos_(ss, hoje, limite) {
  var sheet = ss.getSheetByName(ABA_PROCESSOS);
  var itens = [];
  sheetParaObjetos_(sheet).forEach(function(p) {
    var status = String(p['Status'] || '').toUpperCase();
    if (status !== 'ATIVO' && status !== 'MONITORAR') return;
    var nome = String(p['Nome'] || '').trim();
    if (!nome) return;

    var dataAudi = lfParaData_(p['Data Audiência']);
    var prazoData = lfParaData_(p['prazoData']);
    var diasPrazoRaw = parseInt(p['diasPrazo'], 10) || 0;
    var dataPrazoCliente = lfParaData_(p['dataPrazoCliente']);

    var dias = null, origem = '';
    if (dataAudi) { dias = lfDiasAte_(dataAudi, hoje); origem = 'Audiencia'; }
    else if (prazoData) { dias = lfDiasAte_(prazoData, hoje); origem = 'Prazo processual'; }
    else if (diasPrazoRaw > 0) { dias = diasPrazoRaw; origem = 'Prazo processual'; }
    else if (dataPrazoCliente) { dias = lfDiasAte_(dataPrazoCliente, hoje); origem = 'Providencia do cliente'; }

    if (dias !== null && dias <= limite) {
      itens.push({ titulo: nome, sub: String(p['Número do Processo'] || ''), detalhe: origem, dias: dias });
    }
  });
  itens.sort(function(a, b) { return a.dias - b.dias; });
  return itens;
}

function lfColetarContas_(ss, hoje, limite) {
  var sheet = ss.getSheetByName(ABA_CONTAS);
  var itens = [];
  sheetParaObjetos_(sheet).forEach(function(c) {
    var status = String(c['Status'] || '').toUpperCase();
    if (status.indexOf('PAG') === 0) return;
    var venc = lfParaData_(c['Data Vencimento']);
    if (!venc) return;
    var dias = lfDiasAte_(venc, hoje);
    if (dias !== null && dias <= limite) {
      itens.push({ titulo: String(c['Descricao'] || '(sem descricao)'), sub: String(c['Fornecedor'] || ''), detalhe: c['Valor'] ? 'R$ ' + c['Valor'] : '', dias: dias });
    }
  });
  itens.sort(function(a, b) { return a.dias - b.dias; });
  return itens;
}

function lfColetarHonorarios_(ss, hoje, limite) {
  var sheet = ss.getSheetByName(ABA_HONORARIOS);
  var rows = sheetParaObjetos_(sheet);
  var grupos = {};
  rows.forEach(function(h) {
    var k = (h['Nr Processo'] || '') + '|' + (h['Cliente'] || '');
    (grupos[k] = grupos[k] || []).push(h);
  });

  var itens = [];
  Object.keys(grupos).forEach(function(k) {
    var rs = grupos[k];
    var h0 = rs[rs.length - 1];
    var cliente = h0['Cliente'] || h0['Nr Processo'] || '(sem nome)';
    var proc = h0['Nr Processo'] || '';
    var vTotal = 0, vParc = 0, vEntr = 0, dia = 0, nParc = 0;
    rs.forEach(function(r) {
      vTotal = Math.max(vTotal, lfNbr_(r['Valor Total Contrato']));
      vParc = Math.max(vParc, lfNbr_(r['Valor da Parcela']));
      vEntr = Math.max(vEntr, lfNbr_(r['Valor da Entrada']));
      dia = dia || lfDiaOnly_(r['Dia Vencimento']);
      nParc = nParc || parseInt(r['Número de Parcelas'], 10) || 0;
    });
    if (!vParc && nParc > 0) vParc = Math.max((vTotal - vEntr) / nParc, 0);
    var pags = rs.filter(function(r) { return lfNbr_(r['Valor Recebido']) > 0; });
    var datas = rs.map(function(r) { return lfParaData_(r['Data Pagamento']) || lfParaData_(r['Data Lancamento']); }).filter(Boolean).sort(function(a, b) { return a - b; });
    var start = datas[0] || hoje;

    var plano = [];
    if (vEntr > 0) plano.push({ ref: 'Entrada', key: 'E', valor: vEntr, venc: start });
    if (nParc > 0) {
      var off = (vEntr > 0 || (dia && dia <= start.getDate())) ? 1 : 0;
      for (var i = 1; i <= nParc; i++) {
        plano.push({ ref: lfPad2_(i) + '/' + lfPad2_(nParc), key: i, valor: vParc, venc: lfClampDay_(start.getFullYear(), start.getMonth() + off + (i - 1), dia || start.getDate()) });
      }
    }
    var usados = [];
    plano.forEach(function(p) {
      var m = pags.filter(function(r) { return usados.indexOf(r) < 0 && lfRefKey_(r['Parcela Referencia']) === p.key; })[0];
      if (m) { p.pago = m; usados.push(m); }
    });
    plano.forEach(function(p) {
      if (!p.pago) {
        var m = pags.filter(function(r) { return usados.indexOf(r) < 0 && lfRefKey_(r['Parcela Referencia']) == null; })[0];
        if (m) { p.pago = m; usados.push(m); }
      }
    });

    plano.forEach(function(p) {
      if (p.pago) return;
      var dias = lfDiasAte_(p.venc, hoje);
      if (dias !== null && dias <= limite) {
        itens.push({ titulo: String(cliente), sub: proc ? String(proc) : '', detalhe: 'Parcela ' + p.ref + (p.valor ? ' - R$ ' + p.valor.toFixed(2).replace('.', ',') : ''), dias: dias });
      }
    });
  });
  itens.sort(function(a, b) { return a.dias - b.dias; });
  return itens;
}

// Lembrete de recibo de aluguel (locatário) e informe de repasse (locador) a vencer -
// mesma lógica de "dias até o prazo" das outras seções, usando o mês corrente de
// LOC_LANCAMENTOS e os dias de vencimento/repasse cadastrados no contrato (LOC_CONTRATOS).
function lfColetarRecibosLocPendentes_(ss, hoje, limite) {
  var sheetC = ss.getSheetByName(ABA_LOCC);
  var sheetL = ss.getSheetByName(ABA_LOCL);
  var contratos = sheetParaObjetos_(sheetC).filter(function (c) {
    return String(c['Ativo'] || '').toLowerCase().indexOf('sim') === 0;
  });
  if (!contratos.length) return [];
  var mesRef = lfPad2_(hoje.getMonth() + 1) + '/' + hoje.getFullYear();
  var lancamentos = sheetParaObjetos_(sheetL).filter(function (l) {
    return String(l['Mes Ref'] || '') === mesRef;
  });
  var itens = [];
  contratos.forEach(function (c) {
    var contratoNr = String(c['Contrato'] || '');
    if (!contratoNr) return;
    var lanc = lancamentos.filter(function (l) { return String(l['Contrato'] || '') === contratoNr; })[0];
    if (!lanc) return; // mês ainda não foi gerado pra esse contrato - nada a lembrar
    var locatario = String(c['Locatario'] || '(sem nome)');
    var locador = String(c['Locador'] || '(sem nome)');
    var aluguel = lfNbr_(c['Aluguel']);

    var statusRecibo = String(lanc['Status'] || '').toLowerCase();
    if (statusRecibo.indexOf('receb') !== 0 && statusRecibo.indexOf('pago') !== 0) {
      var diaVenc = lfDiaOnly_(c['Dia Vencimento']);
      if (diaVenc) {
        var dataVenc = lfClampDay_(hoje.getFullYear(), hoje.getMonth(), diaVenc);
        var diasVenc = lfDiasAte_(dataVenc, hoje);
        if (diasVenc !== null && diasVenc <= limite) {
          itens.push({ titulo: 'Recibo — ' + contratoNr + ' (' + locatario + ')', sub: 'Locatário', detalhe: 'Aluguel R$ ' + aluguel.toFixed(2).replace('.', ','), dias: diasVenc });
        }
      }
    }

    var statusRepasse = String(lanc['Status Repasse'] || '').toLowerCase();
    if (statusRepasse.indexOf('repass') !== 0) {
      var diaRep = lfDiaOnly_(c['Dia Repasse']);
      if (diaRep) {
        var dataRep = lfClampDay_(hoje.getFullYear(), hoje.getMonth(), diaRep);
        var diasRep = lfDiasAte_(dataRep, hoje);
        if (diasRep !== null && diasRep <= limite) {
          itens.push({ titulo: 'Informe de repasse — ' + contratoNr + ' (' + locador + ')', sub: 'Locador', detalhe: mesRef, dias: diasRep });
        }
      }
    }
  });
  itens.sort(function (a, b) { return a.dias - b.dias; });
  return itens;
}

// Parcelas de acordo atrasadas/a vencer e repasses ao cliente pendentes além do prazo -
// mesma regra já usada no painel de inadimplência e no card "Acordos" da tela inicial.
function lfColetarAcordosPendentes_(ss, hoje, limite) {
  var sheetC = ss.getSheetByName(ABA_ACC);
  var sheetL = ss.getSheetByName(ABA_ACL);
  var contratos = sheetParaObjetos_(sheetC).filter(function (c) {
    return String(c['Ativo'] || '').toLowerCase().indexOf('sim') === 0;
  });
  var contratosPorId = {};
  contratos.forEach(function (c) { contratosPorId[String(c['ID'])] = c; });
  var lancamentos = sheetParaObjetos_(sheetL).filter(function (l) {
    return contratosPorId.hasOwnProperty(String(l['Contrato'] || ''));
  });
  var itens = [];
  lancamentos.forEach(function (l) {
    var ct = contratosPorId[String(l['Contrato'])] || {};
    var cliente = String(ct['Cliente'] || '(sem nome)');

    if (String(l['Status Recebimento']) !== 'Recebido') {
      var dataPrev = lfParaData_(l['Data Prevista']);
      var atrasadoManual = String(l['Status Recebimento']) === 'Atrasado';
      var dias = dataPrev ? lfDiasAte_(dataPrev, hoje) : null;
      if (atrasadoManual || (dias !== null && dias <= limite)) {
        itens.push({ titulo: 'Parcela de acordo — ' + cliente, sub: String(l['Mes'] || ''), detalhe: 'R$ ' + lfNbr_(l['Valor Previsto']).toFixed(2).replace('.', ','), dias: dias !== null ? dias : -1 });
      }
    }

    if (String(l['Status Recebimento']) === 'Recebido' && String(l['Status Repasse']) !== 'Repassado') {
      var prazoDias = parseInt(ct['Dia Repasse'], 10) || 0;
      var dataReceb = lfParaData_(l['Data Recebimento']);
      if (prazoDias && dataReceb) {
        var diasDesde = Math.round((hoje - dataReceb) / 86400000);
        if (diasDesde > prazoDias) {
          itens.push({ titulo: 'Repasse ao cliente — ' + cliente, sub: 'Recebido em ' + String(l['Data Recebimento'] || ''), detalhe: 'R$ ' + lfNbr_(l['Valor Repasse']).toFixed(2).replace('.', ','), dias: prazoDias - diasDesde });
        }
      }
    }
  });
  itens.sort(function (a, b) { return a.dias - b.dias; });
  return itens;
}

function lfMontarSecaoHtml_(titulo, itens) {
  if (!itens.length) return '';
  var html = '<h3 style="color:#0d1f35;border-bottom:2px solid #c8a96e;padding-bottom:4px;margin-top:22px">' + titulo + '</h3><ul style="padding-left:18px;margin:8px 0">';
  itens.forEach(function(it) {
    html += '<li style="margin-bottom:8px"><strong>' + it.titulo + '</strong>' + (it.sub ? ' - ' + it.sub : '') + '<br>' + '<span style="color:#555">' + (it.detalhe ? it.detalhe + ' - ' : '') + lfLinhaPrazo_(it.dias) + '</span></li>';
  });
  html += '</ul>';
  return html;
}

function enviarAlertasPrazo_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hoje = new Date();
  hoje.setHours(0, 0, 0, 0);
  var limite = LEXFLOW_ALERTA_LIMITE_DIAS;

  var itensProcessos = lfColetarProcessos_(ss, hoje, limite);
  var itensContas = lfColetarContas_(ss, hoje, limite);

  var itensAcordos = lfColetarAcordosPendentes_(ss, hoje, limite);

  var total = itensProcessos.length + itensContas.length + itensHon.length + itensRecibosLoc.length + itensAcordos.length;
  if (total === 0) {
    return { ok: true, enviado: false, motivo: 'nada a alertar' };
  }

  var tz = Session.getScriptTimeZone() || 'America/Sao_Paulo';
  var hojeStr = Utilities.formatDate(hoje, tz, 'dd/MM/yyyy');

  var html = '<div style="font-family:Arial,sans-serif;color:#0d1f35;max-width:600px">'
    + '<h2 style="color:#0d1f35;margin-bottom:2px">Eixo</h2>'
    + '<div style="color:#6b7280;margin-bottom:6px">Alertas de prazo - ' + hojeStr + '</div>'
    + lfMontarSecaoHtml_('Processos', itensProcessos)
    + lfMontarSecaoHtml_('Contas a pagar', itensContas)
    + lfMontarSecaoHtml_('Honorarios a receber', itensHon)
    + lfMontarSecaoHtml_('Locacoes - recibos e informes a enviar', itensRecibosLoc)
    + lfMontarSecaoHtml_('Acordos - parcelas e repasses pendentes', itensAcordos)
    + '</div>';

  MailApp.sendEmail({
    to: LEXFLOW_ALERTA_EMAIL,
    subject: 'Eixo - ' + total + ' alerta(s) de prazo (' + hojeStr + ')',
    htmlBody: html
  });

  return { ok: true, enviado: true, total: total, processos: itensProcessos.length, contas: itensContas.length, honorarios: itensHon.length, locacoes: itensRecibosLoc.length, acordos: itensAcordos.length };
}

// ============================================================
// BACKUP AUTOMATICO DA PLANILHA (copia diaria pro Drive)
// ============================================================
// Protege contra edicao/exclusao acidental na planilha (nao tem "desfazer" facil pra quem
// mexe direto nela) — todo dia de madrugada, guarda uma copia completa numa pasta separada
// do Drive. Mantem só os ultimos LEXFLOW_BACKUP_MAX pra nao acumular arquivo pra sempre.
var LEXFLOW_BACKUP_MAX = 30;

function getOuCriarPastaBackups_() {
  var nome = 'Eixo - Backups';
  var nomeAntigo = 'LexFlow - Backups';
  var it = DriveApp.getRootFolder().getFoldersByName(nome);
  if (it.hasNext()) return it.next();
  // mantém a pasta já existente da marca anterior (LexFlow), se houver, em vez de criar uma nova vazia
  var itAntigo = DriveApp.getRootFolder().getFoldersByName(nomeAntigo);
  if (itAntigo.hasNext()) return itAntigo.next();
  return DriveApp.getRootFolder().createFolder(nome);
}

function lfBackupDiario_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var pasta = getOuCriarPastaBackups_();
  var tz = Session.getScriptTimeZone() || 'America/Sao_Paulo';
  var agora = new Date();
  var nomeBackup = 'Backup Eixo - ' + Utilities.formatDate(agora, tz, 'yyyy-MM-dd HH-mm');
  var arquivoOriginal = DriveApp.getFileById(ss.getId());
  var copia = arquivoOriginal.makeCopy(nomeBackup, pasta);

  // Limpa backups antigos, mantendo só os ultimos LEXFLOW_BACKUP_MAX (os mais recentes).
  var arquivos = pasta.getFilesByType(MimeType.GOOGLE_SHEETS);
  var lista = [];
  while (arquivos.hasNext()) {
    var f = arquivos.next();
    lista.push({ id: f.getId(), data: f.getDateCreated() });
  }
  lista.sort(function (a, b) { return b.data - a.data; });
  var apagados = 0;
  for (var i = LEXFLOW_BACKUP_MAX; i < lista.length; i++) {
    try { DriveApp.getFileById(lista[i].id).setTrashed(true); apagados++; } catch (eDel) {}
  }

  return { ok: true, arquivo: nomeBackup, url: copia.getUrl(), totalBackups: Math.min(lista.length, LEXFLOW_BACKUP_MAX), apagados: apagados };
}

function lfConfigurarTriggerBackup_() {
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === 'lfBackupDiario_') ScriptApp.deleteTrigger(t);
  });
  // Roda de madrugada (3h), fora do horario de uso normal do sistema.
  ScriptApp.newTrigger('lfBackupDiario_').timeBased().everyDays(1).atHour(3).create();
  return { ok: true, mensagem: 'Backup diario configurado (03:00, horario do script).' };
}
